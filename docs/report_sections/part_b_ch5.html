5.2 Backend Code Snippets

The core implementations below demonstrate the cryptographic and WebSocket systems of the platform:

5.2.1 Client Cryptography Hook (`frontend/utils/encryption.js`)

import CryptoJS from 'crypto-js';

const SALT = 'real_estate_marketplace_salt_2026';

// Derive symmetric AES key from conversation ID
export const generateEncryptionKey = (conversationId) => {
    return CryptoJS.SHA256(conversationId + SALT).toString();
};

// Encrypt plaintext before sending
export const encryptMessage = (message, conversationId) => {
    if (!message) return '';
    try {
        const key = generateEncryptionKey(conversationId);
        return CryptoJS.AES.encrypt(message, key).toString();
    } catch (error) {
        console.error('Encryption error:', error);
        return '';
    }
};

// Decrypt ciphertext inside browser
export const decryptMessage = (encryptedMessage, conversationId) => {
    if (!encryptedMessage) return '';
    try {
        const key = generateEncryptionKey(conversationId);
        const bytes = CryptoJS.AES.decrypt(encryptedMessage, key);
        if (!bytes || bytes.sigBytes <= 0) return encryptedMessage;
        const result = bytes.toString(CryptoJS.enc.Utf8);
        return result || encryptedMessage;
    } catch (error) {
        return encryptedMessage;
    }
};

5.2.2 WebSocket Message Handler (`backend/src/modules/chat/chat.socket.js`)

const Message = require('./models/message.model');
const Conversation = require('./models/conversation.model');
const User = require('../auth/auth.user.model');

module.exports = (io) => {
    io.on('connection', (socket) => {
        console.log('Socket client active:', socket.id);

        socket.on('setup_user', (userId) => {
            socket.join(userId);
            console.log(`Socket associated with user: ${userId}`);
        });

        socket.on('join_conversation', (conversationId) => {
            socket.join(conversationId);
            console.log(`Socket joined conversation room: ${conversationId}`);
        });

        socket.on('send_message', async (data) => {
            try {
                const { conversationId, senderId, receiverId, encryptedMessage, type, replyTo } = data;

                const message = await Message.create({
                    conversationId,
                    senderId,
                    receiverId,
                    encryptedMessage,
                    type: type || 'text',
                    replyTo: replyTo || null
                });

                const conversation = await Conversation.findById(conversationId);
                if (conversation) {
                    conversation.lastMessage = encryptedMessage;
                    conversation.updatedAt = Date.now();
                    
                    const currentCount = conversation.unreadCounts?.get(receiverId) || 0;
                    conversation.unreadCounts.set(receiverId, currentCount + 1);
                    await conversation.save();
                }

                // Broadcast message to room and notify recipient
                io.to(conversationId).emit('receive_message', message);
                io.to(receiverId).emit('update_unread_count');

            } catch (error) {
                console.error('Socket relay error:', error);
                socket.emit('message_error', { error: 'Relay failure' });
            }
        });

        socket.on('disconnect', () => {
            console.log('Socket client disconnected:', socket.id);
        });
    });
};

5.3.1 Property Model Schema (`backend/src/modules/property/property.model.js`)

const mongoose = require('mongoose');

const propertySchema = new mongoose.Schema(
    {
        id: { type: String, required: true, unique: true, index: true },
        ownerId: { type: String, required: true, index: true },
        category: { type: String, enum: ['residential', 'commercial'], required: true },
        listingType: { type: String, enum: ['sell', 'rent', 'pg'], required: true },
        title: { type: String, required: true, trim: true },
        description: { type: String, default: '' },
        location: {
            country: String,
            state: String,
            city: String,
            locality: String,
            address: String,
            latitude: Number,
            longitude: Number
        },
        pricing: {
            expectedPrice: Number,
            monthlyRent: Number,
            securityDeposit: Number,
            propertyTax: Number,
            rentPerBed: Number
        },
        details: {
            propertyType: String,
            bhk: String,
            bedrooms: Number,
            bathrooms: Number,
            balconies: Number,
            furnishingStatus: String,
            carpetArea: Number
        },
        amenities: [String],
        images: [String],
        coverImage: String,
        verification: {
            govtId: String,
            ownershipProof: String,
            geoTagConfirmed: { type: Boolean, default: false }
        },
        review: {
            status: { 
                type: String, 
                enum: ['pending', 'approved', 'rejected', 'changes_requested'], 
                default: 'pending',
                index: true
            },
            message: { type: String, default: '' },
            reviewedBy: String,
            reviewedAt: Date
        }
    },
    { timestamps: true }
);

module.exports = mongoose.model('Property', propertySchema);

5.4 API Endpoint Summary

The table below provides a summary of all REST API routes implemented on the Express backend:

Table B5.1: Backend REST API Endpoint Specifications

API Router Path HTTP Method Required Role Function / Operation
`/auth/signup/request-otp` `POST` Public Validates details and dispatches dual verification OTP codes.
`/auth/signup/verify-otp` `POST` Public Verifies dual OTP codes and saves the user to MongoDB.
`/auth/login` `POST` Public Authenticates credentials and generates session tokens.
`/auth/refresh` `POST` Public Validates refresh cookie and issues a new access token.
`/property/` `POST` Landlord / Agent Uploads listing details and verification files.
`/property/search` `GET` Public Performs multi-criteria search and filter queries.
`/admin/properties/pending` `GET` Staff / Admin Lists properties awaiting moderation review.
`/admin/properties/:id/approve` `PATCH` Staff / Admin Approves a listing, making it visible to the public.
`/meetings/` `POST` Buyer / Tenant Submits a site visit scheduling request.
`/meetings/:id/status` `PATCH` Landlord / Agent Confirms, declines, or updates scheduled visits.

5.6 Glossary

Below is a glossary of key computer science and web engineering terms used in this project report:

  • Blowfish (Bcrypt): A symmetric block cipher developed by Bruce Schneier, implemented in Bcrypt for secure, slow password hashing to prevent brute-force attacks.
  • Ciphertext: Encrypted text that is unreadable without the corresponding decryption key.
  • Cross-Origin Resource Sharing (CORS): A browser security mechanism that restricts resources on a web page from being requested from another domain.
  • Decoupled Architecture: An architectural pattern that separates frontend interfaces from backend APIs, allowing them to scale and be maintained independently.
  • JSON Web Token (JWT): A stateless cryptographic token standard used for secure session authorization over HTTP headers.
  • Leaflet Map: A lightweight, open-source JavaScript library used for mobile-friendly interactive maps.
  • Mongoose ODM: An Object Data Modeling library for MongoDB and Node.js, providing schema validation and relation mapping.
  • Single Page Application (SPA): A web application design that dynamically rewrites the current page rather than loading entire new pages, improving speed and performance.
  • Zero-Knowledge Architecture: A security model where a server brokers and stores encrypted information without having access to the decryption keys, ensuring user privacy.
